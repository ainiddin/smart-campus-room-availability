# AGENTS.md

## Project
Smart Campus room availability checker; one room, fictional data.
User story: As a student, I want to check a room's availability so I can choose a valid booking time.

## Acceptance criteria
- AC1: Reject start >= end and past requests.
- AC2: Reject durations longer than 120 minutes.
- AC3: Reject a blocked room.
- AC4: Reject overlaps; allow bookings touching only at endpoints.
- AC5: Do not mutate booking data.
- AC6: Retain clear feedback on desktop and mobile.

## Constraints
No backend, login, database or booking creation.
Do not redesign the UI or add features.

## Structure
- src/lib/availability.ts: pure booking rules
- index.html, src/main.ts, src/style.css: UI
- src/data.ts: demo fixtures
- tests/fixtures.ts: test-only fixtures
- tests/availability.test.ts: tests

## Engineering rules
Prefer pure functions and minimal changes; use at most 2 justified patterns.
Do not mutate booking fixtures; keep responsive UI feedback.
Do not weaken or delete existing test assertions.

## Verification
Run the test script defined in package.json and add focused regression tests.
Report changed files, rationale and test evidence.

===== src/lib/availability.ts =====
// Smart Campus — Room Availability Checker (continued from Practice #06)
// User story: As a student, I want to check a room's availability
// so that I can choose a valid booking time.
// All times are minutes since midnight (0–1440).

export const MAX_DURATION = 120;
export const DAY_END = 1440;

export interface Booking {
  id: string;
  start: number;
  end: number;
}

export interface Room {
  id: string;
  blocked: boolean;
  bookings: Booking[];
}

export type Status = 'Available' | 'Unavailable';

export interface Decision {
  status: Status;
  reason: string;
}

// AC4: intervals are half-open [start, end).
// Touching endpoints (600–660 and 660–720) are allowed.
export function overlaps(aStart: number, aEnd: number, bStart: number, bEnd: number): boolean {
  return aStart < bEnd && bStart < aEnd;
}

export function checkAvailability(room: Room, start: number, end: number, now: number): Decision {
  // AC1: 0 <= start < end <= 1440 and start is after the current time
  if (![start, end, now].every(Number.isInteger)) {
    return { status: 'Unavailable', reason: 'Times must be whole minutes' };
  }
  if (start < 0 || end > DAY_END) {
    return { status: 'Unavailable', reason: 'Times must be between 0 and 1440' };
  }
  if (start >= end) {
    return { status: 'Unavailable', reason: 'Start must be before end' };
  }
  if (start <= now) {
    return { status: 'Unavailable', reason: 'Start must be after the current time' };
  }

  // AC2: maximum duration
  if (end - start > MAX_DURATION) {
    return { status: 'Unavailable', reason: `Maximum duration is ${MAX_DURATION} minutes` };
  }

  // AC3: blocked room
  if (room.blocked) {
    return { status: 'Unavailable', reason: `Room ${room.id} is blocked` };
  }

  // AC4: overlap check (sorted so the earliest conflict is reported)
  const sorted = [...room.bookings].sort((a, b) => a.start - b.start);
  const conflict = sorted.find((b) => overlaps(start, end, b.start, b.end));
  if (conflict) {
    return {
      status: 'Unavailable',
      reason: `Overlaps booking ${conflict.id} (${conflict.start}–${conflict.end})`,
    };
  }

  // AC5
  return { status: 'Available', reason: `Room ${room.id} is free for ${start}–${end}` };
}

===== src/main.ts =====
import { checkAvailability } from './lib/availability';
import { makeRooms } from './data';

const rooms = makeRooms();
const $ = <T extends HTMLElement>(id: string) => document.getElementById(id) as T;

const roomSelect = $<HTMLSelectElement>('room');
roomSelect.innerHTML = rooms
  .map((r) => `<option value="${r.id}">${r.id}${r.blocked ? ' (blocked)' : ''}</option>`)
  .join('');

const current = () => rooms.find((r) => r.id === roomSelect.value)!;

function renderBookings() {
  const r = current();
  $('room-name').textContent = r.id;
  $('bookings').innerHTML = r.bookings.length
    ? r.bookings.map((b) => `<li><b>${b.id}</b> ${b.start}–${b.end}</li>`).join('')
    : '<li class="muted">No bookings</li>';
}

function clearResult() {
  $('result').hidden = true;
}

$('form').addEventListener('submit', (ev) => {
  ev.preventDefault();
  const num = (id: string) => Number(($(id) as HTMLInputElement).value);
  const res = checkAvailability(current(), num('start'), num('end'), num('now'));
  const box = $('result');
  box.hidden = false;
  box.className = 'result ' + (res.status === 'Available' ? 'ok' : 'bad');
  box.innerHTML = `<strong>${res.status}</strong><span>${res.reason}</span>`;
  renderBookings();
});

roomSelect.addEventListener('change', () => { clearResult(); renderBookings(); });
document.querySelectorAll('input').forEach((i) => i.addEventListener('input', clearResult));

renderBookings();

===== tests/availability.test.ts =====
import { describe, it, expect } from 'vitest';
import { checkAvailability, MAX_DURATION } from '../src/lib/availability';
import { makeTestRooms } from './fixtures';

// Fixed clock (minutes since midnight): 540 = 09:00.
const NOW = 540;

// Fresh fixtures in every test — no shared mutable data.
const room = (id: string) => makeTestRooms().find((r) => r.id === id)!;
const check = (id: string, start: number, end: number) =>
  checkAvailability(room(id), start, end, NOW);

describe('AC1 — valid interval after current time', () => {
  it('TC-01 rejects start == end', () => {
    expect(check('C-303', 600, 600).status).toBe('Unavailable');
  });
  it('TC-02 rejects start > end', () => {
    expect(check('C-303', 700, 650).status).toBe('Unavailable');
  });
  it('TC-03 rejects start == now (must be after)', () => {
    expect(check('C-303', 540, 570).status).toBe('Unavailable');
  });
  it('TC-04 rejects a request in the past', () => {
    expect(check('C-303', 480, 510).status).toBe('Unavailable');
  });
  it('TC-05 rejects end > 1440', () => {
    expect(check('C-303', 1400, 1441).status).toBe('Unavailable');
  });
});

describe('AC2 — maximum duration', () => {
  it('TC-06 allows exactly 120 minutes', () => {
    expect(MAX_DURATION).toBe(120);
    expect(check('C-303', 720, 840).status).toBe('Available');
  });
  it('TC-07 rejects 121 minutes', () => {
    expect(check('C-303', 720, 841).status).toBe('Unavailable');
  });
});

describe('AC3 — blocked room', () => {
  it('TC-08 rejects blocked room B-202', () => {
    const r = check('B-202', 660, 720);
    expect(r.status).toBe('Unavailable');
    expect(r.reason).toMatch(/blocked/);
  });
});

describe('AC4 — overlaps, touching allowed', () => {
  it('TC-09 rejects a real overlap (630–690 vs 600–660)', () => {
    const r = check('A-101', 630, 690);
    expect(r.status).toBe('Unavailable');
    expect(r.reason).toMatch(/BK-1/);
  });
  it('TC-10 allows touching end (660–720 after 600–660)', () => {
    expect(check('A-101', 660, 720).status).toBe('Available');
  });
  it('TC-11 allows touching start (720–780 before 780–840)', () => {
    expect(check('A-101', 720, 780).status).toBe('Available');
  });
});

describe('AC5 — decision without changing data', () => {
  it('TC-12 returns Available for a free slot', () => {
    expect(check('A-101', 900, 960).status).toBe('Available');
  });
  it('TC-13 does not modify the room bookings', () => {
    const r = room('A-101');
    const snapshot = structuredClone(r.bookings);
    checkAvailability(r, 900, 960, NOW);
    expect(r.bookings).toEqual(snapshot);
  });
});

describe('regression — rule edges', () => {
  it('AC1: rejects a past request even when the room is otherwise free', () => {
    // 480–510 is before the 09:00 clock; the room has no bookings there.
    expect(check('C-303', 480, 510).status).toBe('Unavailable');
  });

  it('AC1: rejects a reversed interval on a room with bookings', () => {
    // start 900 > end 800 must never be reported as Available.
    const r = check('A-101', 900, 800);
    expect(r.status).toBe('Unavailable');
    expect(r.reason).toMatch(/before end/);
  });

  it('AC2: rejects 121 minutes on a free room', () => {
    expect(check('C-303', 900, 1021).status).toBe('Unavailable');
  });

  it('AC3: rejects a blocked room even for a valid, free slot', () => {
    expect(check('B-202', 900, 960).status).toBe('Unavailable');
  });

  it('AC4: rejects a booking that fully contains an existing booking', () => {
    // 590–700 contains BK-1 (600–660).
    const r = check('A-101', 590, 700);
    expect(r.status).toBe('Unavailable');
    expect(r.reason).toMatch(/BK-1/);
  });

  it('AC4: allows a slot touching existing bookings at both endpoints', () => {
    // 660–780 touches BK-1 at 660 and BK-2 at 780.
    expect(check('A-101', 660, 780).status).toBe('Available');
  });

  it('AC5: an overlap rejection leaves booking order and contents unchanged', () => {
    const r = room('A-101');
    const snapshot = structuredClone(r.bookings);
    checkAvailability(r, 630, 690, NOW);
    expect(r.bookings).toEqual(snapshot);
  });
});
